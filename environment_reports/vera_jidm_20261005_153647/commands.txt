# CUDA validation: exact Python body run via "$LF3R_VERA_JIDM_PYTHON" - <<'PY'.
import subprocess, json
from datetime import datetime
rows=subprocess.check_output(['nvidia-smi','--query-gpu=index,name,memory.free','--format=csv,noheader,nounits'],text=True).strip().splitlines()
print(datetime.now().isoformat(),rows,flush=True)
gpu=max((int(r.rsplit(',',1)[1]),int(r.split(',')[0])) for r in rows)
if gpu[0]<2048:
 raise SystemExit('CUDA check deferred: less than 2048 MiB free')
import torch
print('Torch',torch.__version__,'CUDA runtime',torch.version.cuda,'architectures',torch.cuda.get_arch_list(),flush=True)
device=f'cuda:{gpu[1]}'
assert torch.cuda.get_device_capability(gpu[1])==(12,0)
x=torch.ones((32,32),device=device)
y=x@x
torch.cuda.synchronize(gpu[1])
assert float(y[0,0])==32.0
print('PASS Blackwell CUDA allocation + matmul',device,'peak bytes',torch.cuda.max_memory_allocated(gpu[1]),flush=True)
